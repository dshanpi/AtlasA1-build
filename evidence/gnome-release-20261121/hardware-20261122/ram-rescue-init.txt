#!/rescue/bin/sh
export PATH=/rescue/bin:/usr/bin:/usr/sbin
mkdir -p /proc /sys /dev /run
mount -t proc proc /proc
mount -t sysfs sysfs /sys
mount -t devtmpfs devtmpfs /dev
exec </dev/ttyS0 >/dev/ttyS0 2>&1
export PS1='ATLAS_RAM_RESCUE# '
echo ATLAS_RAM_RESCUE_START
uname -r
for n in $(seq 1 30); do [ -b /dev/mmcblk0 ] && break; sleep 1; done
cat /sys/block/mmcblk0/device/cid
cat /sys/block/mmcblk0/size
cat /proc/mounts
free -m
echo ATLAS_RAM_RESCUE_READY
while :; do /rescue/bin/sh -i; done
